export const SecTOC = ({ items }) => {
  const [active, setActive] = useState(items[0] && items[0].id);
  useEffect(() => {
    if (typeof IntersectionObserver === 'undefined') return;
    const obs = new IntersectionObserver(
      (entries) => entries.forEach(e => { if (e.isIntersecting) setActive(e.target.id); }),
      { rootMargin: '-42% 0px -52% 0px' }
    );
    items.forEach(it => { const el = document.getElementById(it.id); if (el) obs.observe(el); });
    return () => obs.disconnect();
  }, [items]);
  return (
    <nav className="rh-toc">
      <div className="rh-toc__in">
        {items.map(it => (
          <a key={it.id} href={`#${it.id}`} className={active === it.id ? 'is-active' : ''}>{it.label}</a>
        ))}
      </div>
    </nav>
  );
};
