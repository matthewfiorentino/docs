export const LiveUpdates = () => {
  const [items, setItems] = useState([]);
  const [done, setDone] = useState(false);
  const [debug, setDebug] = useState('');

  useEffect(() => {
    fetch('/kb/updates/rss.xml')
      .then(r => {
        if (!r.ok) throw new Error(`HTTP ${r.status} from /kb/updates/rss.xml`);
        return r.text();
      })
      .then(xml => {
        const doc = new DOMParser().parseFromString(xml, 'text/xml');
        const parseErr = doc.querySelector('parsererror');
        if (parseErr) throw new Error('XML parse error: ' + parseErr.textContent.slice(0, 80));
        const els = Array.from(doc.getElementsByTagName('item'));
        if (!els.length) {
          setDebug('Feed loaded but no items found.');
          setDone(true);
          return;
        }
        const CONTENT_NS = 'http://purl.org/rss/1.0/modules/content/';
        const parsed = els.slice(0, 2).map(el => {
          const g = tag => el.getElementsByTagName(tag)[0]?.textContent?.trim() || '';
          const label = g('title');
          const html = el.getElementsByTagNameNS(CONTENT_NS, 'encoded')[0]?.textContent?.trim() || '';
          const inner = new DOMParser().parseFromString(html, 'text/html');
          const headingEl = inner.querySelector('h1,h2,h3,h4,h5,h6');
          const title = headingEl?.textContent?.trim() || label;
          if (headingEl) headingEl.remove();
          const rawDesc = inner.body.textContent.replace(/\s+/g, ' ').trim();
          const desc = rawDesc.length > 160 ? rawDesc.slice(0, 157) + '…' : rawDesc;
          const tag = g('category');
          return { title, desc, date: label };
        }).filter(p => p.title);
        setItems(parsed);
        setDone(true);
      })
      .catch(err => {
        setDebug(err.message);
        setDone(true);
      });
  }, []);

  if (!done) return <div className="ri-updates-loading">Loading…</div>;
  if (debug) return <div className="ri-updates-loading">Updates unavailable — {debug}.</div>;
  if (!items.length) return <a href="/kb/updates" className="ri-updates-more">See all updates →</a>;

  return (
    <div>
      <div className="ri-updates">
        {items.map((item, i) => (
          <UpdateBar
            key={i}
            date={item.date}
            title={item.title}
            desc={item.desc}
          />
        ))}
      </div>
      <a href="/kb/updates" className="ri-updates-more">See all updates →</a>
    </div>
  );
};

## Frequently asked questions

<AccordionGroup>
  <Accordion title="Where do I start if I want to conduct a study at the MUHC?">
    Start with a [feasibility assessment](/kb/feasibility) — before any submission, confirm that your protocol is scientifically and ethically sound, that your team has the required qualifications, and that the facilities and patient population are available. Once ready, submissions go through **Nagano** using the F11 form, which simultaneously triggers REB review and institutional feasibility review. Confirm your Human Research Privileges or Researcher Status before any activities begin.
  </Accordion>
  <Accordion title="What's the difference between the REB and the Personne mandatée?">
    The **Research Ethics Board (REB)** conducts the scientific and ethics review — participant safety, informed consent, risk–benefit ratio, and methodological soundness. REB approval is valid for one year and must be renewed annually.

    The **Personne mandatée** is the MUHC institutional authority who issues the final authorization to conduct research. This authorization is granted only after all reviews are complete — REB approval, feasibility reviews, and any required Health Canada approvals. Conditional REB approval is not sufficient. The two roles are independent — REB approval does not automatically confer MUHC authorization.
  </Accordion>
  <Accordion title="Does my study need Health Canada approval?">
    It depends on whether your study involves an investigational product. **Investigational drug or biologic** → Clinical Trial Application (CTA), No Objection Letter (NOL). **Medical device** (Class II, III, or IV) → Investigational Testing Authorization (ITA). **Natural health product** → NHP CTA, Notice of Authorization (NOA).

    Observational studies and studies using biological specimens only — with no investigational product — do not require a Health Canada submission. When in doubt, contact the Research Facilitator early in your planning process.
  </Accordion>
  <Accordion title="What training do I need to join a study team?">
    Training requirements depend on study type. All research staff must complete the **RI-MUHC SOP Reader** (tracked acknowledgement of each applicable SOP via TalentLMS) and the **Competency Assessment** (renewed every 2 years for interventional studies, every 5 years for observational or retrospective work).

    For drug studies (Level I), **ICH GCP** and **Health Canada Division 5** training are also required. Device studies (Level II) require GCP and **ISO 14155**. No team member may perform any study task before their training is documented and they appear on the Task Delegation Log. See [Training Requirements](/training/compliance-requirements).
  </Accordion>
</AccordionGroup>
