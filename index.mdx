import { useState, useEffect, useRef } from 'react'

export const StatCounter = ({ value, prefix = '', suffix = '', duration = 1400 }) => {
  const ref = useRef();
  const [current, setCurrent] = useState(0);
  useEffect(() => {
    if (!ref.current) return;
    let isVisible = false;
    const io = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting && !isVisible) {
        isVisible = true;
        setCurrent(0);
        const start = performance.now();
        const step = (now) => {
          const t = Math.min((now - start) / duration, 1);
          setCurrent(value * (1 - Math.pow(1 - t, 3)));
          if (t < 1) requestAnimationFrame(step);
        };
        requestAnimationFrame(step);
      } else if (!entry.isIntersecting) {
        isVisible = false;
      }
    }, { threshold: 0.4 });
    io.observe(ref.current);
    return () => io.disconnect();
  }, [value]);
  const decimals = value % 1 !== 0 ? 1 : 0;
  return <span ref={ref}>{prefix}{current.toFixed(decimals)}{suffix}</span>;
};

export const LiveUpdates = () => {
  const [items, setItems] = useState([]);
  const [done, setDone] = useState(false);
  const [debug, setDebug] = useState('');

  useEffect(() => {
    fetch('/kb/updates/rss.xml')
      .then(r => {
        if (!r.ok) throw new Error(`HTTP ${r.status} from /kb/updates/rss.xml`);
        return r.text();
      })
      .then(xml => {
        const doc = new DOMParser().parseFromString(xml, 'text/xml');
        const parseErr = doc.querySelector('parsererror');
        if (parseErr) throw new Error('XML parse error: ' + parseErr.textContent.slice(0, 80));
        const els = Array.from(doc.getElementsByTagName('item'));
        if (!els.length) {
          setDebug('Feed loaded but no items found.');
          setDone(true);
          return;
        }
        const CONTENT_NS = 'http://purl.org/rss/1.0/modules/content/';
        const parsed = els.slice(0, 2).map(el => {
          const g = tag => el.getElementsByTagName(tag)[0]?.textContent?.trim() || '';
          const label = g('title');
          const html = el.getElementsByTagNameNS(CONTENT_NS, 'encoded')[0]?.textContent?.trim() || '';
          const inner = new DOMParser().parseFromString(html, 'text/html');
          const headingEl = inner.querySelector('h1,h2,h3,h4,h5,h6');
          const title = headingEl?.textContent?.trim() || label;
          if (headingEl) headingEl.remove();
          const rawDesc = inner.body.textContent.replace(/\s+/g, ' ').trim();
          const desc = rawDesc.length > 160 ? rawDesc.slice(0, 157) + '…' : rawDesc;
          const tag = g('category');
          return { title, desc, date: label };
        }).filter(p => p.title);
        setItems(parsed);
        setDone(true);
      })
      .catch(err => {
        setDebug(err.message);
        setDone(true);
      });
  }, []);

  if (!done) return <div className="ri-updates-loading">Loading…</div>;
  if (debug) return <div className="ri-updates-loading">Updates unavailable — {debug}.</div>;
  if (!items.length) return <a href="/kb/updates" className="ri-updates-more">See all updates →</a>;

  return (
    <div>
      <div className="ri-updates">
        {items.map((item, i) => (
          <UpdateBar
            key={i}
            date={item.date}
            title={item.title}
            desc={item.desc}
          />
        ))}
      </div>
      <a href="/kb/updates" className="ri-updates-more">See all updates →</a>
    </div>
  );
};

export const RoleExplorer = () => {
  const [role, setRole] = useState('pi');
  const containerRef = useRef(null);
  const active = ROLES.find(r => r.id === role);

  const handleRoleSwitch = (id) => {
    setRole(id);
    if (containerRef.current) {
      const rect = containerRef.current.getBoundingClientRect();
      if (rect.top < 80) {
        containerRef.current.scrollIntoView({ behavior: 'smooth', block: 'start' });
      }
    }
  };

  return (
    <div className="ri-sh" ref={containerRef}>
      <div className="ri-sh-tabs-wrap">
        <div className="ri-sh-tabs">
          {ROLES.map(r => (
            <button
              key={r.id}
              className={`ri-sh-tab${role === r.id ? ' ri-sh-tab--active' : ''}`}
              onClick={() => handleRoleSwitch(r.id)}

              <Icon icon={r.icon} className="ri-sh-tab-icon" />
