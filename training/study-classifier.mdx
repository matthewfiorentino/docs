export const Classifier = () => {
  const [currentQ, setCurrentQ]             = useState(1);
  const [answers, setAnswers]               = useState({});
  const [history, setHistory]               = useState([]);
  const [resType, setResType]               = useState(null);
  const [intType, setIntType]               = useState(null);
  const [done, setDone]                     = useState(false);
  const [outOfScope, setOutOfScope]         = useState(false);
  const [pendingCallout, setPendingCallout] = useState(null);
  const [stepNum, setStepNum]               = useState(1);

  const getNextKey = (qKey, val) => {
    if (qKey === 1)       return val === 'yes' ? 2 : 'no-human';
    if (qKey === 2) {
      if (val === 'interventional')           return '3int';
      if (val === 'observational-prospective') return '3obs';
      return '3retro';
    }
    if (qKey === '3int')  return val === 'drug' ? '4drug' : 5;
    if (qKey === '4drug') return 5;
    if (qKey === '3obs')  return 5;
    if (qKey === '3retro')return 5;
    if (qKey === 5)       return val === 'yes' ? 6 : 8;
    if (qKey === 6)       return 7;
    if (qKey === 7)       return 8;
    if (qKey === 8)       return 'result';
    return 'result';
  };

  const handleAnswer = (qKey, val) => {
    const newAnswers = { ...answers, [qKey]: val };
    const newHistory = [...history, qKey];
    let newResType = resType;
    let newIntType = intType;
    if (qKey === 2)      newResType = val;
    if (qKey === '3int') newIntType = val;

    const qdef = QUESTIONS[qKey];
    let callout = null;
    if (qdef && qdef.calloutOnYes    && val === 'yes')    callout = qdef.calloutOnYes;
    if (qdef && qdef.calloutOnWaiver && val === 'waiver') callout = qdef.calloutOnWaiver;

    const nextKey = getNextKey(qKey, val);

    setAnswers(newAnswers);
    setHistory(newHistory);
    setResType(newResType);
    setIntType(newIntType);
    setPendingCallout(callout);

    if (nextKey === 'no-human') {
      setOutOfScope(true);
    } else if (nextKey === 'result') {
      setDone(true);
      setStepNum(8);
    } else {
      setCurrentQ(nextKey);
      setStepNum(Q_STEP[nextKey] || stepNum);
    }
  };

  const handleReset = () => {
    setCurrentQ(1);
    setAnswers({});
    setHistory([]);
    setResType(null);
    setIntType(null);
    setDone(false);
    setOutOfScope(false);
    setPendingCallout(null);
    setStepNum(1);
  };

  const getLevel = () => {
    if (resType === 'interventional') {
      if (intType === 'drug')   return 1;
      if (intType === 'device') return 2;
      if (intType === 'nhp')    return 3;
      if (intType === 'low')    return 4;
    }
    if (resType === 'observational-prospective') return 4;
    if (resType === 'retrospective')             return 5;
    return null;
  };

  const getFlags = () => {
    const f = [];
    if (answers[5] === 'yes')
      f.push({ t: 'amber', l: 'Personal Health Information — LSSSS / Loi 25', x: 'Your study involves identifiable PHI. Ensure your REB application includes a data management and privacy protection plan. Loi 25 requires disclosure of any confidentiality breach within 72 hours to the CAI and affected individuals.' });
    if (answers[6] === 'waiver')
      f.push({ t: 'amber', l: 'Consent waiver — TCPS2 Art. 3.7 / 5.5A', x: 'The REB must determine the research could not practicably proceed with consent, benefits outweigh risks, and privacy is protected. If using RAMQ or provincial registry data, a separate MSSS/CAI authorization is required — typically 3–6 months.' });
    if (answers[7] === 'yes')
      f.push({ t: 'coral', l: 'Cross-border data transfer — ÉFVP required', x: 'Your study will share identifiable or coded data outside RI-MUHC/MUHC. A Privacy Impact Assessment (ÉFVP) must be completed before any transfer occurs. Address this in your REB submission and data transfer agreements.' });
    const vuln = answers[8];
    if (vuln === 'minor' || vuln === 'both')
      f.push({ t: 'sky', l: 'Minors — Civil Code of Québec, Art. 21', x: 'Research involving minors requires written parental or guardian consent. Assent should be obtained from children aged 7–13 years as developmentally appropriate. Research posing more than minimal risk requires special justification under TCPS2 Section 4.' });
    if (vuln === 'incapable' || vuln === 'both')
      f.push({ t: 'coral', l: 'Adults unable to consent — Personne mandétée / Civil Code, Art. 21', x: 'Research involving adults who cannot consent requires authorization from a Personne mandétée (mandatary), tutor, or curator. The MUHC Authorization process applies. See [SOP-CR-008](/sops/cr-008).' });
    if (resType === 'interventional' && (intType === 'drug' || intType === 'device' || intType === 'nhp'))
      f.push({ t: 'teal', l: 'ICH E6(R3) — effective April 2026', x: 'Health Canada formally adopted ICH E6(R3) effective April 1, 2026. The guideline introduces a QMS approach and updates requirements for electronic systems and risk-proportionate monitoring.' });
    if (resType === 'retrospective' && answers['3retro'] === 'identifiable')
      f.push({ t: 'amber', l: 'RAMQ / registry data — MSSS / CAI authorization', x: 'Secondary use of RAMQ administrative data or provincial registries requires a formal authorization request to MSSS and/or CAI, plus an ÉFVP. Separate from REB approval — typically 3–6 months.' });
    return f;
  };

  const RcCallout = ({ c }) => {
    if (!c) return null;
    return <div className={'rc-callout ' + c.type} dangerouslySetInnerHTML={{ __html: c.text }} />;
  };

  const qdef = !outOfScope && !done ? QUESTIONS[currentQ] : null;

  const renderResult = () => {
    const level = getLevel();
    if (!level || !RESULTS[level]) {
      return (
        <div className="rc-fallback">
          <strong>Unable to determine training level.</strong> Contact QA: <strong>qaclinicalresearch@muhc.mcgill.ca</strong>
        </div>
      );
    }
    const r = RESULTS[level];
    const flags = getFlags();
    const reqs = r.reqs(answers['4drug'] === 'yes');
    return (
      <div className="rc-result">
        <div className="rc-result-head">
          <span className="rc-level-badge">{r.badge}</span>
          <span className="rc-result-title">{r.title}</span>
        </div>
        <div className="rc-result-section">
          <div className="rc-result-section-label">Required — complete before study activation</div>
          <div dangerouslySetInnerHTML={{ __html: reqs }} />
        </div>
        {r.renewal && (
          <div className="rc-result-section">
            <div className="rc-result-section-label">Renewal cycle</div>
            <div dangerouslySetInnerHTML={{ __html: r.renewal }} />
          </div>
        )}
        <div className="rc-result-section">
          <div className="rc-result-section-label">Regulatory framework</div>
          <div className="rc-reg-context" dangerouslySetInnerHTML={{ __html: r.framework }} />
        </div>
        {flags.length > 0 && (
          <div className="rc-result-section">
            <div className="rc-result-section-label">Flags and additional considerations</div>
            <div className="rc-flags">
              {flags.map((flag, i) => (
                <div key={i} className={'rc-flag ' + flag.t}>
                  <strong>{flag.l}</strong>
                  {flag.x}
                </div>
              ))}
            </div>
          </div>
        )}
      </div>
    );
  };

  return (
    <div className="rc-wrap">

      <div className="rc-reg-note">
        <strong>Regulatory basis:</strong> Training requirements derive from <a href="/sops/overview">[SOP-CR-003](/sops/cr-003)</a>, <strong>ICH E6(R3)</strong> (Health Canada, effective Apr 2026), <strong>Food and Drug Regulations Part C Div.&#8201;5</strong> (GUI-0100), <strong>ISO 14155:2020</strong>, <strong>TCPS2 (2022)</strong>, and Quebec&#8217;s <strong>LSSSS</strong>. Training level is based on your highest-risk study across all active protocols.
      </div>

      <div className="rc-progress">
        {[1,2,3,4,5,6,7,8].map(i => (
          <div key={i} className={'rc-prog-dot' + ((done && i <= 8) || i < stepNum ? ' done' : i === stepNum && !done ? ' active' : '')} />
        ))}
        <span className="rc-prog-label">{done ? 'Complete' : 'Step ' + stepNum + ' of 8'}</span>
      </div>

      {history.length > 0 && (
        <div className="rc-history">
          {history.map((qKey, idx) => {
            const qd = QUESTIONS[qKey];
            if (!qd) return null;
            const labelMap = Q_LABELS[qKey] || {};
            const aLabel = labelMap[answers[qKey]] || String(answers[qKey]);
            return (
              <div key={idx} className="rc-hist-item">
                <div className="rc-hist-num">&#10003;</div>
                <div>
                  <div className="rc-hist-q">{qd.q}</div>
                  <div className="rc-hist-a">{aLabel}</div>
                </div>
              </div>
            );
          })}
        </div>
      )}

      {pendingCallout && <RcCallout c={pendingCallout} />}

      {qdef && (
        <div className="rc-q-block">
          {qdef.part && <div className="rc-part-label">{qdef.part}</div>}
          <div className="rc-q-row">
            <div className="rc-q-num pass">&#8250;</div>
            <div>
              <div className="rc-q-main">{qdef.q}</div>
              {qdef.hint && <div className="rc-q-hint">{qdef.hint}</div>}
            </div>
          </div>

          {qdef.type === 'yn' ? (
            <div className="rc-answers">
              <button className="rc-btn" onClick={() => handleAnswer(currentQ, 'yes')}>
                {(qdef.ynLabels && qdef.ynLabels[0]) || 'Yes'}
              </button>
              <button className="rc-btn" onClick={() => handleAnswer(currentQ, 'no')}>
                {(qdef.ynLabels && qdef.ynLabels[1]) || 'No'}
              </button>
            </div>
          ) : (
            <div className="rc-options">
              {qdef.opts.map(opt => (
                <button key={opt.val} className="rc-opt-btn" onClick={() => handleAnswer(currentQ, opt.val)}>
                  {opt.label}
                  {opt.desc && <span className="rc-opt-desc">{opt.desc}</span>}
                </button>
              ))}
            </div>
          )}

          {qdef.callout && <RcCallout c={qdef.callout} />}
        </div>
      )}

      {outOfScope && (
        <div className="rc-no-human">
          <strong>Outside scope of this tool.</strong> Research that does not involve human participants, biological material, or personal data does not require RI-MUHC human research training. Contact QA if you are uncertain: <strong>qaclinicalresearch@muhc.mcgill.ca</strong>
        </div>
      )}

      {done && renderResult()}

      <div className="rc-footer">
        <button className="rc-reset-btn" onClick={handleReset}>&#8635; Start over</button>
        <p className="rc-disclaimer">This tool provides guidance based on <a href="/sops/overview">[SOP-CR-003](/sops/cr-003)</a>. Training level is based on your <strong>highest-risk active study</strong>. Results do not constitute formal compliance advice. When in doubt, contact QA.</p>
      </div>

    </div>
  );
};
