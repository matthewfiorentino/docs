import { useEffect } from 'react'

<Tip>
  **Before you start:** make sure your project is research and not a QA/QI initiative. The [Is It Research?](/kb/is-it-research) screening tool decides this in a few minutes and routes you to the right next step. QA/QI projects are exempt from REB review under TCPS2 Article 2.5 and follow a different path.
</Tip>

export const StudyRoadmap = () => {
  useEffect(() => {
    let alive = true;
    fetch('/roadmap-script.js')
      .then(r => r.text())
      .then(code => { if (alive) (new Function(code))(); });
    return () => { alive = false; };
  }, []);
  return <div id="rm-root" />;
};
