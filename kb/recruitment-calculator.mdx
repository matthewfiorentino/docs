import { useEffect } from 'react'

export const RecruitmentCalculator = () => {
  useEffect(() => {
    const existing = document.getElementById('rc-script');
    if (existing) existing.remove();
    const script = document.createElement('script');
    script.id = 'rc-script';
    script.src = '/recruitment-calc-script.js';
    document.body.appendChild(script);
    return () => {
      const s = document.getElementById('rc-script');
      if (s) s.remove();
    };
  }, []);

  return <div id="rc-mount" />;
};

<RecruitmentCalculator />
